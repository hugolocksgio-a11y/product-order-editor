    <script>
        // --- 1. SYSTEM STATE ---
        let systemData = {
            lockdownMode: false,
            databaseText: "No information currently in database."
        };

        // Master User List (Hugo is hardcoded as the ultimate admin)
        let users = [
            { 
                username: "hugo.giovannangelo", 
                password: "091116", 
                role: "admin", 
                canEdit: true, 
                allowedInLockdown: true 
            }
        ];

        let currentUser = null;

        // --- 2. LOGIN LOGIC ---
        function attemptLogin() {
            const userIn = document.getElementById("username").value;
            const passIn = document.getElementById("password").value;
            const errorMsg = document.getElementById("error-msg");

            // Find user
            const user = users.find(u => u.username === userIn && u.password === passIn);

            if (!user) {
                errorMsg.innerText = "Incorrect username or password.";
                return;
            }

            // Check Lockdown status
            if (systemData.lockdownMode && !user.allowedInLockdown) {
                errorMsg.innerText = "ACCESS DENIED: System is in Lockdown.";
                return;
            }

            // Success
            currentUser = user;
            document.getElementById("username").value = "";
            document.getElementById("password").value = "";
            errorMsg.innerText = "";
            
            showDashboard();
        }

        function logout() {
            currentUser = null;
            document.getElementById("laptop-container").style.display = "flex";
            document.getElementById("dashboard").style.display = "none";
        }

        // --- 3. DASHBOARD & UI LOGIC ---
        function showDashboard() {
            document.getElementById("laptop-container").style.display = "none";
            document.getElementById("dashboard").style.display = "flex";
            
            document.getElementById("welcome-msg").innerText = "Logged in as: " + currentUser.username;
            
            // Database text setup
            const dbContent = document.getElementById("db-content");
            dbContent.value = systemData.databaseText;
            
            if (currentUser.canEdit) {
                dbContent.removeAttribute("readonly");
                document.getElementById("save-db-btn").style.display = "inline-block";
            } else {
                dbContent.setAttribute("readonly", true);
                document.getElementById("save-db-btn").style.display = "none";
            }

            // Lockdown Banner
            document.getElementById("lockdown-banner").style.display = systemData.lockdownMode ? "block" : "none";

            // Admin Panel setup
            if (currentUser.username === "hugo.giovannangelo") {
                document.getElementById("admin-panel").style.display = "block";
                updateAdminPanel();
            } else {
                document.getElementById("admin-panel").style.display = "none";
            }
        }

        function saveDatabase() {
            systemData.databaseText = document.getElementById("db-content").value;
            const msg = document.getElementById("save-msg");
            msg.innerText = "Saved!";
            setTimeout(() => { msg.innerText = ""; }, 2000);
        }

        // --- 4. ADMIN CONTROLS (Hugo Only) ---
        function toggleLockdown() {
            systemData.lockdownMode = !systemData.lockdownMode;
            updateAdminPanel();
            document.getElementById("lockdown-banner").style.display = systemData.lockdownMode ? "block" : "none";
        }

        function addUser() {
            const newU = document.getElementById("new-user").value;
            const newP = document.getElementById("new-pass").value;
            const canEdit = document.getElementById("can-edit").checked;
            const canLock = document.getElementById("can-lockdown").checked;

            if(newU === "" || newP === "") {
                alert("Username and password cannot be empty!");
                return;
            }

            // Check if exists
            if(users.some(u => u.username === newU)) {
                alert("User already exists!");
                return;
            }

            users.push({
                username: newU,
                password: newP,
                role: "user",
                canEdit: canEdit,
                allowedInLockdown: canLock
            });

            // Clear inputs
            document.getElementById("new-user").value = "";
            document.getElementById("new-pass").value = "";
            document.getElementById("can-edit").checked = false;
            document.getElementById("can-lockdown").checked = false;

            updateAdminPanel();
        }

        function removeUser(username) {
            if (username === "hugo.giovannangelo") {
                alert("You cannot remove the master admin!");
                return;
            }
            users = users.filter(u => u.username !== username);
            updateAdminPanel();
        }

        function updateAdminPanel() {
            // Update lockdown button text
            const lockBtn = document.getElementById("lockdown-btn");
            if (systemData.lockdownMode) {
                lockBtn.innerText = "Disable Lockdown";
                lockBtn.classList.add("danger-btn");
            } else {
                lockBtn.innerText = "Enable Lockdown";
                lockBtn.classList.remove("danger-btn");
            }

            // Render users table
            const tbody = document.getElementById("users-list");
            tbody.innerHTML = "";

            users.forEach(u => {
                let removeBtnHtml = u.username === "hugo.giovannangelo" 
                    ? `<span style="color:#7f8c8d;">Master Admin</span>` 
                    : `<button class="btn danger-btn" style="padding: 5px 10px;" onclick="removeUser('${u.username}')">Remove</button>`;

                let row = `<tr>
                    <td>${u.username}</td>
                    <td>${u.role}</td>
                    <td>${u.canEdit ? "✅ Yes" : "❌ No"}</td>
                    <td>${u.allowedInLockdown ? "✅ Yes" : "❌ No"}</td>
                    <td>${removeBtnHtml}</td>
                </tr>`;
                tbody.innerHTML += row;
            });
        }
